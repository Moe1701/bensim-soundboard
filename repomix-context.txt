```javascript
const { initOscClient } = require('./src/osc/x32Client');

const oscClient = initOscClient();

const CHANNELS = [
  '/ch/01', '/ch/02', '/ch/03', '/ch/04', '/ch/05',
  '/ch/06', '/ch/07', '/ch/08', '/ch/09', '/ch/10',
  '/auxin/01'
];
const BUSES = ['01', '02', '03'];

// Hilfsfunktion zum Pausieren
const sleep = (ms) => new Promise(resolve => setTimeout(resolve, ms));

async function runTest() {
  console.log('\n🚀 Starte X32 Hardware-Test in 2 Sekunden...');
  await sleep(2000);

  for (const bus of BUSES) {
    console.log(`\n====================================`);
    console.log(`📡 TESTE BUS ${bus} (Instruktor ${parseInt(bus)})`);
    console.log(`====================================`);
    
    for (const ch of CHANNELS) {
      const address = `${ch}/mix/${bus}/level`;
      
      console.log(`➡️ ${address} -> 100%`);
      oscClient.setFader(address, 100);
      
      await sleep(300); // Fader 300ms oben halten
      
      oscClient.setFader(address, 0);
      await sleep(100); // 100ms Pause bis zum nächsten Kanal
    }
  }
  
  console.log('\n✅ Test-Routine erfolgreich beendet!');
  console.log('Du kannst das Skript jetzt mit Strg+C abbrechen.');
  process.exit(0);
}

// Wir geben dem UDP-Port 500ms Zeit, sich zu öffnen, bevor der Test startet
```

## File: backend/src/osc/x32Client.js
```javascript
const osc = require('osc');

// ACHTUNG: Die IP-Adresse deines echten X32 Mischpults im Simulationszentrum.
// Für die Offline-Entwicklung ist das gerade ein Dummy-Wert.
const X32_IP = '192.168.1.100'; 
const X32_PORT = 10024; // Standard-Port für X32 OSC

let udpPort;

function initOscClient() {
  udpPort = new osc.UDPPort({
    localAddress: "0.0.0.0", // Lauscht auf allen lokalen Netzwerk-Adaptern
    localPort: 53000,        // Freier Port auf unserem Linux-Backend
    remoteAddress: X32_IP,
    remotePort: X32_PORT,
    metadata: true
  });

  udpPort.on("error", (err) => {
    console.error("⚠️ OSC Fehler:", err.message);
  });

  udpPort.on("ready", () => {
    console.log(`🎛️  OSC Client bereit. Verbunden mit X32 auf ${X32_IP}:${X32_PORT}`);
  });

  // UDP-Port öffnen
  udpPort.open();

  // Wir geben ein Objekt mit Hilfsfunktionen zurück, 
  // die der WebSocket-Handler später einfach aufrufen kann.
  return {
    // Schaltet einen Kanal stumm oder an
    setMute: (channelAddress, isMuted) => {
      // X32 Logik: 0 = Mute (Ton aus), 1 = On (Ton an)
      const value = isMuted ? 0 : 1; 
      
      udpPort.send({
        address: channelAddress, // z.B. /ch/01/mix/on
        args: [
          { type: "i", value: value }
        ]
      });
      console.log(`OSC -> ${channelAddress} = ${value} (Int)`);
    },

    // Setzt die Fader-Lautstärke
    setFader: (channelAddress, levelPercent) => {
      // Das Tablet sendet 0-100%, das X32 erwartet Float von 0.0 bis 1.0
      const oscLevel = levelPercent / 100;
      
      udpPort.send({
        address: channelAddress,
        args: [
          { type: "f", value: oscLevel }
        ]
      });
      // console.log-Spam deaktiviert für flüssigere Terminal-Ausgabe
      // console.log(`OSC -> ${channelAddress} = ${oscLevel} (Float)`);
    }
  };
}

module.exports = { initOscClient };
```

## File: backend/src/websocket/socketHandler.js
```javascript
// Speichert alle aktiven Verbindungen (welches Tablet hat welche Rolle)
const connectedClients = new Map(); 

// --- X32 ROUTING MAPS ---
const ROLE_TO_BUS = {
  'instructor-1': '01', // Bus 1
  'instructor-2': '02', // Bus 2
  'instructor-3': '03'  // Bus 3
};

const CHANNEL_MAP = {
  'tn1': '/ch/01', 'tn2': '/ch/02', 'tn3': '/ch/03', 'tn4': '/ch/04',
  'tn5': '/ch/05', 'tn6': '/ch/06', 'tn7': '/ch/07', 'tn8': '/ch/08',
  'raumL': '/ch/09', 'raumR': '/ch/10', 'aux1': '/auxin/01'
};

function setupWebSocket(ws, oscClient) {
  ws.on('message', (message) => {
    try {
      const data = JSON.parse(message);

      if (data.type === 'REGISTER_ROLE') {
        connectedClients.set(ws, data.role);
        console.log(`✅ Registriert: ${data.role}. Aktive Tablets: ${connectedClients.size}`);
        return;
      }

      if (data.type === 'AUDIO_UPDATE') {
        const role = connectedClients.get(ws);
        if (!role) return; 

        const { target, id, property, value, route, active } = data.payload;
        const targetBus = ROLE_TO_BUS[role];
        
        if (!targetBus) return;

        // -- X32 OSC MAPPING --
        if (target === 'channel') {
          const basePath = CHANNEL_MAP[id];
          if (!basePath) return;

          if (property === 'mute') {
            oscClient.setMute(`${basePath}/mix/${targetBus}/on`, value);
          } 
          else if (property === 'volume') {
            oscClient.setFader(`${basePath}/mix/${targetBus}/level`, value);
          }
          // NEU: Die Solo-Logik
          else if (property === 'solo') {
            console.log(`🎧 Solo-Modus [${role}] für ${id} -> ${value ? 'AN' : 'AUS'}`);
            
            // Wir gehen blitzschnell alle 11 Hardware-Kanäle durch
            for (const [chanId, chanPath] of Object.entries(CHANNEL_MAP)) {
              if (chanId !== id) {
                // Wenn Solo aktiviert wird (value=true) -> Alle anderen stummschalten (Mute=true)
                // Wenn Solo deaktiviert wird (value=false) -> Alle anderen wieder anmachen (Mute=false)
                oscClient.setMute(`${chanPath}/mix/${targetBus}/on`, value);
              } else if (value === true) {
                // Den Solo-Kanal selbst zur Sicherheit immer hörbar machen (Mute=false)
                oscClient.setMute(`${chanPath}/mix/${targetBus}/on`, false);
              }
            }
          }
        }

        if (target === 'talkback') {
          // Platzhalter für den nächsten Schritt!
          console.log(`🎙️ Talkback Routing [${route}] -> ${active ? 'AN' : 'AUS'}`);
        }
      }

    } catch (error) {
      console.error('⚠️ Fehler beim Verarbeiten der WebSocket-Nachricht:', error);
    }
  });

  ws.on('close', () => {
    const role = connectedClients.get(ws);
    if (role) {
      connectedClients.delete(ws);
      console.log(`❌ Verbindung getrennt: ${role}`);
    }
  });
}

module.exports = { setupWebSocket };
```

```javascript
const express = require('express');
const http = require('http');
const WebSocket = require('ws');
const path = require('path');
const os = require('os');
const qrcode = require('qrcode-terminal'); // NEU: QR Code Generator
const { setupWebSocket } = require('./src/websocket/socketHandler');
const { initOscClient } = require('./src/osc/x32Client');

const PORT = 8080;

// Hilfsfunktion: Findet die WLAN/LAN IP-Adresse deines PCs
function getLocalIp() {
  const interfaces = os.networkInterfaces();
  for (const name of Object.keys(interfaces)) {
    for (const iface of interfaces[name]) {
      if (iface.family === 'IPv4' && !iface.internal) {
        return iface.address;
      }
    }
  }
  return 'localhost';
}

const localIp = getLocalIp();
const tabletUrl = `http://${localIp}:${PORT}`;

console.log('====================================');
console.log('🎙️ BenSim Soundboard Backend startet');
console.log('====================================');

const app = express();
const server = http.createServer(app);

const frontendPath = path.join(__dirname, '../frontend/dist');
app.use(express.static(frontendPath));

app.use((req, res) => {
  res.sendFile(path.join(frontendPath, 'index.html'));
});

const oscClient = initOscClient();
const wss = new WebSocket.Server({ server });

wss.on('connection', (ws) => {
  console.log('Neu: Tablet (oder Host) hat sich verbunden.');
  setupWebSocket(ws, oscClient);
});

server.listen(PORT, '0.0.0.0', () => {
  console.log(`✅ Server laufen synchron auf Port ${PORT}.`);
  console.log(`🖥️  Host-PC: http://localhost:${PORT}`);
  console.log(`📱 Tablets: ${tabletUrl}`);
  
  // NEU: QR-Code im Terminal ausgeben
  console.log('\n--- SCAN MICH FÜR TABLET-ZUGANG ---');
  qrcode.generate(tabletUrl, { small: true });
  console.log('====================================\n');
});
```

## File: frontend/src/App.tsx
```typescript
import { useState } from 'react';
import type { InstructorRole } from './types/audio';
import { Header } from './components/Header';
import { MonitoringBoard } from './components/MonitoringBoard';
import { TalkbackMatrix } from './components/TalkbackMatrix';
import { RoleSelector } from './components/RoleSelector';
import { useOscConnection } from './hooks/useOscConnection';

export default function App() {
  const [activeRole, setActiveRole] = useState<InstructorRole | null>(null);

  const { isConnected, sendAudioUpdate } = useOscConnection(
    activeRole as InstructorRole
  );

  if (!activeRole) {
    return <RoleSelector onSelectRole={setActiveRole} />;
  }

  return (
    <div className="min-h-screen bg-slate-50 text-slate-800 p-4 md:p-8 flex flex-col font-sans select-none overflow-hidden gap-6">
      <Header role={activeRole} isConnected={isConnected} />
      <MonitoringBoard onAudioUpdate={sendAudioUpdate} />
      <TalkbackMatrix onAudioUpdate={sendAudioUpdate} />
    </div>
  );
}
```
