```javascript
const osc = require('osc');

// ACHTUNG: Die IP-Adresse deines echten X32 Mischpults im Simulationszentrum.
// Für die Offline-Entwicklung ist das gerade ein Dummy-Wert.
const X32_IP = '192.168.1.100'; 
const X32_PORT = 10024; // Standard-Port für X32 OSC

let udpPort;

function initOscClient() {
  udpPort = new osc.UDPPort({
    localAddress: "0.0.0.0", 
    localPort: 0,            // <--- ÄNDERUNG: 0 wählt dynamisch einen freien Port
    remoteAddress: X32_IP,
    remotePort: X32_PORT,
    metadata: true
  });

  udpPort.on("error", (err) => {
    console.error("⚠️ OSC Fehler:", err.message);
  });

  udpPort.on("ready", () => {
    console.log(`🎛️  OSC Client bereit. Verbunden mit X32 auf ${X32_IP}:${X32_PORT}`);
  });

  // UDP-Port öffnen
  udpPort.open();

  // Wir geben ein Objekt mit Hilfsfunktionen zurück, 
  // die der WebSocket-Handler später einfach aufrufen kann.
  return {
    // Schaltet einen Kanal stumm oder an
    setMute: (channelAddress, isMuted) => {
      // X32 Logik: 0 = Mute (Ton aus), 1 = On (Ton an)
      const value = isMuted ? 0 : 1; 
      
      udpPort.send({
        address: channelAddress, // z.B. /ch/01/mix/on
        args: [
          { type: "i", value: value }
        ]
      });
      console.log(`OSC -> ${channelAddress} = ${value} (Int)`);
    },

    // Setzt die Fader-Lautstärke
    setFader: (channelAddress, levelPercent) => {
      // Das Tablet sendet 0-100%, das X32 erwartet Float von 0.0 bis 1.0
      const oscLevel = levelPercent / 100;
      
      udpPort.send({
        address: channelAddress,
        args: [
          { type: "f", value: oscLevel }
        ]
      });
      // console.log-Spam deaktiviert für flüssigere Terminal-Ausgabe
      // console.log(`OSC -> ${channelAddress} = ${oscLevel} (Float)`);
    }
  };
}

module.exports = { initOscClient };
```

```typescript
import { useState, useEffect, useRef, useCallback } from 'react';
import type { InstructorRole } from '../types/audio';

export function useOscConnection(role: InstructorRole | null) {
  const [isConnected, setIsConnected] = useState(false);
  const wsRef = useRef<WebSocket | null>(null);

  useEffect(() => {
    if (!role) {
      return;
    }

    const wsUrl = `ws://${window.location.hostname}:8080`;
    const ws = new WebSocket(wsUrl);
    wsRef.current = ws;

    ws.onopen = () => {
      setIsConnected(true);
      ws.send(JSON.stringify({ type: 'REGISTER_ROLE', role }));
    };

    ws.onclose = () => {
      setIsConnected(false);
    };

    return () => {
      if (ws.readyState === WebSocket.OPEN || ws.readyState === WebSocket.CONNECTING) {
        ws.close();
      }
    };
  }, [role]);

  const sendAudioUpdate = useCallback((update: any) => {
    if (wsRef.current && wsRef.current.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify({ type: 'AUDIO_UPDATE', ...update }));
    }
  }, []);

  const claimVideoMaster = useCallback(() => {
    if (wsRef.current && wsRef.current.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify({ type: 'CLAIM_VIDEO_MASTER', role }));
    }
  }, [role]);

  return { isConnected, sendAudioUpdate, claimVideoMaster };
}
```

```javascript
const connectedClients = new Map(); 
let currentVideoMaster = null; // Speichert, welche Rolle gerade den Video-Mix diktiert

  'room': '14',  // Mixbus 14 (Raumklang)
  'pat1': '11',  // Mixbus 11 (In-Ear Patient 1)
  'pat2': '12',  // Mixbus 12 (In-Ear Patient 2)
  'tn': '15'     // Mixbus 15 (Alle TN - falls in Zukunft genutzt)
};

function setupWebSocket(ws, oscClient, wss) {
  ws.on('message', (message) => {
    try {
      const data = JSON.parse(message);

      if (data.type === 'REGISTER_ROLE') {
        connectedClients.set(ws, data.role);
        console.log(`✅ Registriert: ${data.role}. Aktive Tablets: ${connectedClients.size}`);
        return;
      }
      
      if (data.type === 'CLAIM_VIDEO_MASTER') {
        currentVideoMaster = data.role;
        console.log(`🎥 Video-Regie übernommen durch: ${data.role}`);
        // Informiere alle Tablets, wer gerade Video-Master ist
        wss.clients.forEach(client => {
          if (client.readyState === 1) { // WebSocket.OPEN
            client.send(JSON.stringify({ type: 'VIDEO_MASTER_UPDATE', masterRole: currentVideoMaster }));
          }
        });
        return;
      }

      if (data.type === 'AUDIO_UPDATE') {
        const role = connectedClients.get(ws);
        if (!role) return; 

        const { target, id, property, value, route, active } = data.payload;
        
        // -- ZONE A (MASTER): Gesamtlautstärke des In-Ear Busses --
        if (target === 'inear_master') {
          const targetBus = ROLE_TO_BUS[role];
          if (!targetBus) return;
          
          if (property === 'volume') {
            // Regelt den Bus-Masterfader (z.B. /bus/01/mix/fader)
            oscClient.setFader(`/bus/${targetBus}/mix/fader`, value);
          } else if (property === 'mute') {
            oscClient.setMute(`/bus/${targetBus}/mix/on`, value);
          }
        }

        // -- ZONE A (CHANNELS): MONITORING MIX --
        if (target === 'channel') {
          const targetBus = ROLE_TO_BUS[role];
          if (!targetBus) return;
          
          const basePath = CHANNEL_MAP[id];
          if (!basePath) return;

          if (property === 'mute') {
            // Mute im eigenen In-Ear Send
            oscClient.setMute(`${basePath}/mix/${targetBus}/on`, value);
            // Wenn Video-Master, auch im Main-Mix muten
            if (role === currentVideoMaster) {
              oscClient.setMute(`${basePath}/mix/on`, value);
            }
          } 
          else if (property === 'volume') {
            // Fader für eigenen In-Ear Send
            oscClient.setFader(`${basePath}/mix/${targetBus}/level`, value);
            // Wenn Video-Master, Fader-Bewegung auch auf den X32 Main-Fader spiegeln
            if (role === currentVideoMaster) {
              oscClient.setFader(`${basePath}/mix/fader`, value);
            }
          }
          else if (property === 'solo') {
            console.log(`🎧 Solo-Modus [${role}] für ${id} -> ${value ? 'AN' : 'AUS'}`);
            
            for (const [chanId, chanPath] of Object.entries(CHANNEL_MAP)) {
              if (chanId !== id) {
                oscClient.setMute(`${chanPath}/mix/${targetBus}/on`, value);
              } else if (value === true) {
                oscClient.setMute(`${chanPath}/mix/${targetBus}/on`, false);
              }
            }
          }
        }

        // -- ZONE B: MASTER OUTPUTS (RAUM & SCHAUSPIELER) --
        if (target === 'master') {
          const basePath = MASTER_MAP[id];
          if (!basePath) return;

          // Bei Bussen und Main LR heißt der Fader-Pfad '/mix/fader' (nicht '/mix/level' wie bei Sends)
          if (property === 'mute') {
            oscClient.setMute(`${basePath}/mix/on`, value);
            console.log(`🎚️ Master Mute -> ${id} = ${value ? 'STUMM' : 'AN'}`);
          } 
          else if (property === 'volume') {
            oscClient.setFader(`${basePath}/mix/fader`, value);
          }
        }

        // -- ZONE C: TALKBACK MATRIX (PTT) --
        if (target === 'talkback') {
          console.log(`🎙️ Talkback [${role}] -> Route: ${route}, Status: ${active ? 'AN' : 'AUS'}`);
          
          const micChannel = INSTRUCTOR_MICS[role];
          const destBus = TALKBACK_DESTINATIONS[route];

          if (micChannel && destBus) {
            oscClient.setMute(`${micChannel}/mix/${destBus}/on`, !active); 
          } else {
            console.warn(`⚠️ Talkback-Ziel '${route}' oder Mikrofon für '${role}' nicht in der Map definiert!`);
          }
        }
      }

    } catch (error) {
      console.error('⚠️ Fehler beim Verarbeiten der WebSocket-Nachricht:', error);
    }
  });

  ws.on('close', () => {
    const role = connectedClients.get(ws);
    if (role) {
      connectedClients.delete(ws);
      console.log(`❌ Verbindung getrennt: ${role}`);
    }
  });
}
```

```typescript
  const [activeRole, setActiveRole] = useState<InstructorRole | null>(null);

  const { isConnected, sendAudioUpdate, claimVideoMaster } = useOscConnection(
    activeRole as InstructorRole
  );

  if (!activeRole) {
    return <RoleSelector onSelectRole={setActiveRole} />;
  }

  return (
    <div className="h-screen w-screen bg-slate-50 text-slate-800 p-2 md:p-3 flex flex-col font-sans select-none overflow-hidden gap-2">
      {/* Der Kopf-Bereich (Rolle, Status) */}
      <Header role={activeRole} isConnected={isConnected} onClaimVideoMaster={claimVideoMaster} />

      {/* OBERE HÄLFTE: Zone A (In-Ear) */}
      <div className="flex-[3] min-h-0">
        <MonitoringBoard onAudioUpdate={sendAudioUpdate} />
      </div>

      {/* UNTERE HÄLFTE: Zone B & C (Raum & PTT) */}
      <div className="flex-[2] min-h-0">
        <RoomControl onAudioUpdate={sendAudioUpdate} />
      </div>
    </div>
  );
}
```
